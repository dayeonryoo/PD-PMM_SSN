cmake_minimum_required(VERSION 3.16)

project(KSP_QP VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)

if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# -march=native tunes for the CPU that runs *this* build, so binaries are not portable to
# other machines (they can crash with "Illegal instruction" on a different/older CPU). Safe as
# long as everyone builds from source on their own machine; do not reuse a compiled build/
# across machines, bake one into a Docker image that runs elsewhere, or ship it as a release
# binary without switching this to a portable baseline (e.g. -march=x86-64-v2).
add_compile_options(-O3 -march=native -funroll-loops)

set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Dependencies
# ----------------------
include(FetchContent)

# Eigen
FetchContent_Declare(
  eigen
  URL https://gitlab.com/libeigen/eigen/-/archive/5.0.0/eigen-5.0.0.zip
)
FetchContent_MakeAvailable(eigen)

# METIS (nested-dissection ordering candidate for ordering_select.hpp) + its dependency GKlib.
# Both fetched directly from KarypisLab's GitHub (Apache-2.0 / Apache-2.0+LGPL-2.1-or-later) to
# avoid older, non-commercially-licensed METIS tarballs still floating around in system/conda
# packages. Off switch provided since the integration below works around METIS's build system
# expecting a pre-`make config` tree (see comments below); ordering_select.hpp/ksp_qp.tpp must
# build correctly with this OFF, falling back to Eigen's native AMD/Natural orderings only.
option(KSP_QP_BUILD_METIS "Fetch and build METIS from GitHub for a 3rd (nested-dissection) ordering candidate" ON)
if (KSP_QP_BUILD_METIS)
  # GKlib has no clean release tags; pinned to the master HEAD commit as of 2026-08-31.
  FetchContent_Declare(
    gklib
    URL https://github.com/KarypisLab/GKlib/archive/3b7d61b9f885063c89901f3901fb4426f9cfb58f.zip
  )
  FetchContent_MakeAvailable(gklib)

  # METIS's top-level CMakeLists.txt (unlike GKlib's) predates modern target-based CMake: it
  # locates GKlib's headers via a directory-scoped `include_directories(${GKLIB_PATH}/include)`
  # rather than linking a GKlib CMake target, and normally expects `GKLIB_PATH` to be set by its
  # own `make config` step. Point it at our live GKlib checkout directly -- no install step
  # needed, since GKlib's headers already live under <source>/include.
  set(GKLIB_PATH ${gklib_SOURCE_DIR} CACHE PATH "GKlib root for METIS's build" FORCE)

  FetchContent_Declare(
    metis
    URL https://github.com/KarypisLab/METIS/archive/refs/tags/v5.2.1.zip
  )
  FetchContent_GetProperties(metis)
  if (NOT metis_POPULATED)
    FetchContent_Populate(metis)
    # METIS's top-level CMakeLists.txt unconditionally add_subdirectory()s "build/xinclude",
    # which does not exist in a raw checkout -- it's normally generated by `make config`
    # (see METIS's own Makefile). Replicate that step: write IDXTYPEWIDTH/REALTYPEWIDTH (required
    # -- include/metis.h has no #else default) ahead of the real header, pinned to 32 to match
    # Eigen's default 32-bit StorageIndex, since Eigen::MetisOrdering passes StorageIndex buffers
    # straight to METIS's idx_t* parameters with no size check (see ordering_select.hpp).
    file(MAKE_DIRECTORY ${metis_SOURCE_DIR}/build/xinclude)
    file(READ ${metis_SOURCE_DIR}/include/metis.h _metis_header)
    file(WRITE ${metis_SOURCE_DIR}/build/xinclude/metis.h
      "#define IDXTYPEWIDTH 32\n#define REALTYPEWIDTH 32\n${_metis_header}")
    file(COPY ${metis_SOURCE_DIR}/include/CMakeLists.txt DESTINATION ${metis_SOURCE_DIR}/build/xinclude)
    add_subdirectory(${metis_SOURCE_DIR} ${metis_BINARY_DIR})
  endif()
endif()

# GoogleTest (only needed when tests are built)
option(KSP_QP_BUILD_TESTS "Build the gtest test suite" ON)
if (KSP_QP_BUILD_TESTS)
  FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.zip
  )
  # Match the rest of the project against the parent compiler/runtime settings.
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
endif()

# ----------------------
# Solver library
# ----------------------
add_library(ksp_qp INTERFACE)
target_include_directories(ksp_qp INTERFACE include)
# Eigen's unsupported modules (including MINRES in unsupported/Eigen/IterativeSolvers)
# are bundled in the Eigen download above. This explicit include ensures the path resolves.
target_include_directories(ksp_qp INTERFACE $<BUILD_INTERFACE:${eigen_SOURCE_DIR}>)
target_link_libraries(ksp_qp
    INTERFACE
        Eigen3::Eigen)

if (KSP_QP_BUILD_METIS)
  # METIS's own headers are exposed only via directory-scoped include_directories(), never a
  # target -- and it defines no find_package config -- so wire both up manually. GKlib::GKlib
  # provides its own transitive include dir via target_include_directories, so only metis's own
  # (generated) xinclude dir needs to be added explicitly here.
  target_include_directories(ksp_qp INTERFACE ${metis_SOURCE_DIR}/build/xinclude)
  target_link_libraries(ksp_qp INTERFACE metis GKlib::GKlib)
  target_compile_definitions(ksp_qp INTERFACE KSP_QP_HAVE_METIS)
endif()
